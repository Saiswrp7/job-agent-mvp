% Built to Interview Prep/Resume/Resume_Formatting_Preferences.md
% (Charter, 10.5pt, 0.45in margins, 6pt between sections, 1.5pt itemsep)
%
% ONE deliberate change from the existing .tex files: the header is a single
% column, not tabularx. A two-column header is the most common reason an ATS
% interleaves a resume — the parser reads across the page and mixes the two
% columns together. render.py asserts the extracted text order, so if this ever
% regresses the build fails rather than shipping a scrambled parse.
\documentclass[10.5pt,a4paper]{article}

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[margin=0.45in]{geometry}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage[hidelinks]{hyperref}
\usepackage{charter}

\titleformat{\section}{\normalsize\bfseries\uppercase}{}{0em}{}[\titlerule]
\titlespacing*{\section}{0pt}{6pt}{5pt}

\pagestyle{empty}
\setlength{\parskip}{0pt}
\setlength{\parindent}{0pt}
\setlist[itemize]{itemsep=1.5pt, parsep=0pt, topsep=2pt, leftmargin=12pt}

\newcommand{\role}[3]{%
  \textbf{#1} \hfill \textbf{#2}\\[1pt]%
  \textit{#3}\vspace{3pt}%
}

\begin{document}

{\Large\bfseries ((NAME))}\\[2pt]
{\small ((CONTACT))}
\vspace{2pt}

\section*{Summary}
\vspace{2pt}
((SUMMARY))

\section*{Experience}
\vspace{2pt}
((EXPERIENCE))

\section*{Education}
\vspace{2pt}
((EDUCATION))

\section*{Skills}
\vspace{2pt}
((SKILLS))

\end{document}
